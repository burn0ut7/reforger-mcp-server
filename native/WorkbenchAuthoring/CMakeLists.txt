cmake_minimum_required(VERSION 3.24)
project(WorkbenchAuthoring VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Workbench authoring requires Windows x64 and MSVC.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Network)

add_library(rstauthoring MODULE Plugin.cpp Compatibility.cpp PropertyEditors.cpp PropertySource.cpp EditableChoices.cpp ChoicesClient.cpp ArraySelection.cpp MenuIcons.cpp)
target_link_libraries(rstauthoring PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network iphlpapi)
target_compile_options(rstauthoring PRIVATE /W4 /WX /permissive- /Zc:__cplusplus /Brepro)
target_link_options(rstauthoring PRIVATE /Brepro)
target_compile_definitions(rstauthoring PRIVATE QT_NO_DEBUG)
set_target_properties(rstauthoring PROPERTIES PREFIX "")
add_custom_command(TARGET rstauthoring POST_BUILD
    COMMAND "${CMAKE_LINKER}" /dump /dependents "/out:${CMAKE_BINARY_DIR}/dependencies.txt" $<TARGET_FILE:rstauthoring>
    VERBATIM)
install(TARGETS rstauthoring RUNTIME DESTINATION generic LIBRARY DESTINATION generic)

include(CTest)
if(BUILD_TESTING)
    add_executable(authoring_array_command_test NativeArrayCommandsTest.cpp Compatibility.cpp)
    target_link_libraries(authoring_array_command_test PRIVATE Qt6::Core)
    target_compile_options(authoring_array_command_test PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    add_test(NAME typed_array_command COMMAND authoring_array_command_test)
    add_executable(authoring_source_text_test SourceTextTest.cpp)
    target_link_libraries(authoring_source_text_test PRIVATE Qt6::Core)
    target_compile_options(authoring_source_text_test PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    add_test(NAME source_text COMMAND authoring_source_text_test)
    add_executable(authoring_compatibility_test CompatibilityTest.cpp Compatibility.cpp)
    target_link_libraries(authoring_compatibility_test PRIVATE Qt6::Core)
    target_compile_options(authoring_compatibility_test PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    add_test(NAME compatibility COMMAND authoring_compatibility_test)
    add_executable(authoring_load_test PluginLoadTest.cpp)
    target_link_libraries(authoring_load_test PRIVATE Qt6::Core Qt6::Gui)
    target_compile_options(authoring_load_test PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    add_test(NAME exact_dll_refusal COMMAND authoring_load_test $<TARGET_FILE:rstauthoring>)
    add_test(NAME qt_plugin_refusal COMMAND authoring_load_test $<TARGET_FILE:rstauthoring> generic)
    add_test(NAME choices_config_refusal COMMAND authoring_load_test $<TARGET_FILE:rstauthoring> config)
    add_executable(authoring_choices_test EditableChoicesTest.cpp EditableChoices.cpp)
    target_link_libraries(authoring_choices_test PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)

    add_test(NAME editable_choices COMMAND authoring_choices_test)

    add_executable(authoring_client_test ChoicesClientTest.cpp ChoicesClient.cpp)

    add_test(NAME choices_client COMMAND authoring_client_test)
endif()
